import { Aside } from '@astrojs/starlight/components';
import Command from '$lib/components/Command.astro';
import ParamTable from '$lib/components/ParamTable.astro';

<Aside>
Every command on this page requires the `Manage Members` permission.
</Aside>

## /warn add
<Command>
    <div slot="usage">
        This command takes in a user and a reason and gives the user a warning.
    </div>
    <div slot="params">
        <ParamTable input={{
            user: {
                description: "The user to warn",
                type: "User",
                required: true
            },
            reason: {
                description: "The reason for the warning",
                type: "String",
                required: true
            }
        }} />
    </div>
</Command>

## /warn remove
<Command>
    <div slot="usage">
        This command takes a warning ID and removes it.
    </div>
    <div slot="params">
        <ParamTable input={{
            id: {
                description: "The ID of the warning",
                type: "String",
                required: true
            }
        }} />
    </div>
</Command>

## /warn list
<Command>
    <div slot="usage">
        You can use this command to list out all the warnings in the server or just the warnings for a specific user.
    </div>
    <div slot="params">
        <ParamTable input={{
            user: {
                description: "The user to list warnings for (shows all users if unspecified)",
                type: "User",
                required: false
            }
        }} />
    </div>
</Command>
